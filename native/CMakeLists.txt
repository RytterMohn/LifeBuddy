cmake_minimum_required(VERSION 3.22)
project(mobile_agent_jni LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(NOT DEFINED LLAMA_SOURCE_DIR)
    set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama-build)
add_library(mobile_agent_jni SHARED llama_jni.cpp)
set_target_properties(mobile_agent_jni PROPERTIES OUTPUT_NAME llama)
target_link_libraries(mobile_agent_jni PRIVATE llama android log)
